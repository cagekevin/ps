## 源码

```js
{
  props: {
    title: {
      type: String,
      default: "设计助理"
    },
    showOk: {
      type: Boolean,
      default: true
    },
    okText: {
      type: String,
      default: ""
    }
  },
  data: function () {
    return {
      isShow: false,
      resolve: "",
      reject: "",
      promise: ""
    };
  },
  methods: {
    cancel: function () {
      this.initVal();
      this.reject("cancel");
    },
    confirm: function () {
      this.resolve("confirm");
      this.initVal();
    },
    show: function () {
      var t = this;
      this.isShow = true;
      this.promise = new Promise(function (e, n) {
        t.resolve = e;
        t.reject = n;
      });
      return this.promise;
    },
    initVal: function () {
      this.isShow = false;
      this.title = "设计助理";
      this.showOk = true;
      this.okText = "";
    }
  },
  watch: {
    isShow: function () {
      if (this.isShow == 0) {
        this.initVal();
      }
    }
  }
}
```
