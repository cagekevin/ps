{
  components: {
    Icon: ie.a
  },
  props: {
    setFolderIconInfo: {
      type: Object,
      default: function () {
        return {
          icon: 0,
          color: 0,
          offsetHeight: 0
        };
      }
    }
  },
  data: function () {
    return {
      currentColor: 0,
      currentIcon: 0,
      iconGroupList: [this.getNumberArray(0, 15), this.getNumberArray(16, 31), this.getNumberArray(32, 50)],
      blockColor: ["#D3D3D3", "#F7BD21", "#EB5E22", "#CA49F5", "#F4A989", "#3FB799", "#3498FF", "#7EDF6E"]
    };
  },
  mounted: function () {
    this.currentColor = this.setFolderIconInfo.color;
    this.currentIcon = this.setFolderIconInfo.icon;
  },
  methods: {
    getNumberArray: function (t, e) {
      var n = [];
      for (var a = t; a <= e; a++) {
        n.push(a);
      }
      return n;
    },
    setBlockColor: function (t) {
      this.currentColor = t;
      this.confirmFun();
    },
    setBlockIcon: function (t) {
      this.currentIcon = t;
      this.confirmFun();
    },
    confirmFun: function () {
      this.$emit("confirmFun", this.currentColor, this.currentIcon);
    },
    close: function () {
      this.$emit("closeFileSvg");
    }
  }
}
