### 3.2 `src/cep/eval-script.ts`

要点：

```ts
export function evalScript<T = string>(script: string, timeout = 30000): Promise<T> {
  return new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error('evalScript 超时')), timeout)
    window.__adobe_cep__.evalScript(script, (res: string) => {
      clearTimeout(timer)
      if (res === 'EvalScript error.') return reject(new Error('EvalScript error'))
      resolve(res as T)
    })
  })
}
```

**验收**：能在浏览器 console 里 mock 测试

### 3.3 `src/cep/ps.ts`

要点：

```ts
const ENV = { VUE_APP_USER_DIR: '/SheJiZhuLi', NODE_ENV: 'production', ... }

let inited = false

export async function initJsx(): Promise<void> {
  if (inited) return
  const ext = window.__adobe_cep__.getSystemPath('extension')
  const p = ext.replace(/\\/g, '/') + '/lib/all.decrypted.js'
  await evalScript(`$.evalFile("${p}")`)
  await evalScript(`Sys_setSystemEnvInfo('${JSON.stringify(ENV)}')`)
  inited = true
}

export async function getFontList(): Promise<FontItem[]> {
  await initJsx()
  const raw = await evalScript<string>('ztgl_initSystemList()')
  return JSON.parse(raw)
}
```

**验收**

- [ ] `initJsx()` 只执行一次（有 `inited` 标记）
- [ ] `getFontList()` 返回数组

## Step 5 · 第一个页面：字体（2 小时）

**文件**：`src/pages/font/index.vue`

要点：

```vue
<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { getFontList } from '@/cep/ps'
import type { FontItem } from '@/cep/types'

const fonts = ref<FontItem[]>([])
const loading = ref(false)
const error = ref('')

onMounted(async () => {
  loading.value = true
  try {
    fonts.value = await getFontList()
  } catch (e: any) {
    error.value = e.message
  } finally {
    loading.value = false
  }
})
</script>

<template>
  <div class="font-page">
    <div v-if="loading">加载中...</div>
    <div v-else-if="error" class="error">{{ error }}</div>
    <ul v-else>
      <li v-for="f in fonts" :key="f.family">{{ f.family }}</li>
    </ul>
  </div>
</template>
```

**验收**

- [ ] 打开面板能看到 259 个字体名
- [ ] 没有控制台报错

## 常见问题

| 现象 | 原因 | 解决 |
|---|---|---|
| 白屏 | `base` 不是 `./` | 改 vite.config |
| 菜单里没有 | manifest 或调试模式 | 检查 bundle id、PlayerDebugMode |
| `ztgl_initSystemList` 不是函数 | 没加载 jsx | 检查 `initJsx()` |
| `undefined 不是对象` | 没设环境 | 检查 `Sys_setSystemEnvInfo` |
| 字体列表空 | jsx 返回格式不对 | 打印原始返回 |
| 构建后找不到 lib | 静态资源没复制 | 加 deploy 脚本 |
